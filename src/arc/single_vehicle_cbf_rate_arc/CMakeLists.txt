# ---- CbfDiagnostics: side-channel telemetry for visualization / experiment
# logging (see EXPERIMENT_GUI_PLAN_20260829.md). Generated here rather than

# libtorch (client_lib's penn_gamma_selector target) and OSQP (FetchContent)
# both live outside the standard system library search path -- embed an
# RPATH into the installed executable so it finds them without needing
# LD_LIBRARY_PATH set at launch.
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)

# The offline diagnostic / label-export executables link ${TORCH_LIBRARIES}
# directly AND call into rclcpp in their own translation units. libtorch's
# imported target propagates -D_GLIBCXX_USE_CXX11_ABI to whatever links it,
# so with a pre-CXX11-ABI libtorch wheel those TUs compile old-ABI and fail
# to resolve rclcpp's new-ABI std::string symbols (e.g. rclcpp::get_logger).
# The deployed node is unaffected (it reaches torch only through
# penn_gamma_selector's PRIVATE link, fully encapsulated). Default ON so
# run_full_validation.sh etc. are unchanged; set OFF to build just the
# node + tests against such a libtorch (e.g. for a SITL experiment).
option(SV_CBF_BUILD_DIAG_TOOLS
       "Build the offline diagnostic/label-export tools (need a CXX11-ABI libtorch)" ON)

# Systematic C++-vs-Python parity check -- see
# client_lib/tools/penn_gamma_parity_check.cpp. Run after any change to
# penn_gamma_selector.cpp or a checkpoint swap, against a batch of states
# generated by penn_gat_training/diag_gen_cpp_parity_states.py.
#
# Torch found/linked directly here (not inherited from penn_gamma_selector)
# because this tool reads the states .pt file itself via
# torch::jit::pickle_load, unlike penn_gamma_diagnostic.cpp which only ever
# touches the selector's opaque PIMPL interface and needs no torch headers
# of its own.
find_package(Torch REQUIRED)

if(SV_CBF_BUILD_DIAG_TOOLS)
add_executable(penn_gamma_parity_check
               client_lib/tools/penn_gamma_parity_check.cpp)

target_link_libraries(penn_gamma_parity_check
  penn_gamma_selector
  ${TORCH_LIBRARIES}
)
endif()
