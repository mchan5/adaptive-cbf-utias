set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# libtorch discovery. scripts/build.sh resolves the prefix from the project
# venv and passes it as -DARC_TORCH_CMAKE_PREFIX_PATH, so the rest of the
# workspace builds under the system python (which has catkin_pkg/empy/lark).
# Falls back to a $VIRTUAL_ENV probe for a hand-run `colcon build` with the
# venv active.
if(NOT ARC_TORCH_CMAKE_PREFIX_PATH AND DEFINED ENV{VIRTUAL_ENV})
  execute_process(
    COMMAND "$ENV{VIRTUAL_ENV}/bin/python3" -c "import torch; print(torch.utils.cmake_prefix_path, end='')"
    OUTPUT_VARIABLE ARC_TORCH_CMAKE_PREFIX_PATH
    RESULT_VARIABLE _torch_probe_result
  )
  if(NOT _torch_probe_result EQUAL 0)
    set(ARC_TORCH_CMAKE_PREFIX_PATH "")
  endif()
endif()
if(NOT ARC_TORCH_CMAKE_PREFIX_PATH)
  message(FATAL_ERROR
    "single_vehicle_cbf_rate_arc needs libtorch. Build with scripts/build.sh, "
    "or pass -DARC_TORCH_CMAKE_PREFIX_PATH=<torch/share/cmake>, or set "
    "VIRTUAL_ENV to a python env that has torch.")
endif()
list(APPEND CMAKE_PREFIX_PATH ${ARC_TORCH_CMAKE_PREFIX_PATH})
find_package(Torch REQUIRED)

# ---- OSQP (v1 C API), fetched and built from source -- no system/Python-wheel
# osqp exposes a linkable C library or CMake config. ----
include(FetchContent)
set(OSQP_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(OSQP_BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(OSQP_BUILD_DEMO_EXE OFF CACHE BOOL "" FORCE)
set(OSQP_BUILD_UNITTESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(
  osqp
  GIT_REPOSITORY https://github.com/osqp/osqp.git
  GIT_TAG v1.0.0
)
FetchContent_MakeAvailable(osqp)

find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)

# ---- PENN+GAT inference, isolated into its own target ----------------------

add_library(penn_gamma_selector STATIC
  src/penn_gamma_selector.cpp
)
target_include_directories(penn_gamma_selector PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(penn_gamma_selector
  PRIVATE ${TORCH_LIBRARIES}
  PUBLIC Eigen3::Eigen
)

# ---- Vehicle-agnostic CBF-QP + PENN/GAT orchestration core -----------------
# No px4_msgs dependency -- this is what a second vehicle backend (e.g.
# Crazyflie) links against too, instead of duplicating any of this logic.
# nav_msgs/visualization_msgs are here because both vehicle backends consume
# the exact same state_estimator/local_position/odom and /arc/obstacles
# message contracts, not because this is PX4-specific.

add_library(rate_autopilot_core STATIC
  src/rate_autopilot_core.cpp
  src/body_rate_hocbf.cpp
  src/entry_certification_gate.cpp
  src/body_rate_qp_osqp.cpp
)

target_include_directories(rate_autopilot_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)

target_link_libraries(rate_autopilot_core
  penn_gamma_selector
  osqpstatic
  Eigen3::Eigen
)

ament_target_dependencies(rate_autopilot_core
  rclcpp
  nav_msgs
  visualization_msgs
  Eigen3
)

# ---- PX4-specific I/O around rate_autopilot_core ---------------------------

add_library(single_vehicle_cbf_rate_client STATIC

)

target_include_directories(single_vehicle_cbf_rate_client PUBLIC

target_link_libraries(single_vehicle_cbf_rate_client

# CbfDiagnostics is generated by this same package (rosidl_generate_interfaces
# in the parent CMakeLists, which runs before this add_subdirectory). Link the
# C++ typesupport so single_vehicle_cbf_rate_client.cpp can publish it.

  ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(single_vehicle_cbf_rate_client "${cpp_typesupport_target}")
